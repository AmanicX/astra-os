# Intel CPU/GPU stack
# Selected by installer/lib/hw.sh when the CPU vendor is GenuineIntel
# and/or the GPU is Intel.

intel-ucode                  # Intel CPU microcode
mesa                         # Mesa graphics; provides libva-mesa-driver in the 2026 snapshot
vulkan-intel                 # ANV Vulkan driver
lib32-mesa                   # 32-bit Mesa for Steam/Wine
lib32-vulkan-intel           # 32-bit ANV
intel-media-driver           # VA-API for Broadwell and newer
libva                        # VA-API client library
libva-utils                  # vainfo diagnostics
libvpl                       # Intel Video Processing Library
vpl-gpu-rt                   # oneVPL runtime for Tiger Lake and newer

# The installer adds libva-intel-driver only for GPU generations that need the
# legacy G45/HD driver. It is kept in the offline repo for that case.
