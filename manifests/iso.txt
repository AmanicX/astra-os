# ISO-only tools
# These are present in the live environment but are not installed on the
# target by default. Target packages come from base.txt, desktop.txt, apps.txt,
# drivers-*.txt, fonts-themes.txt and astra.txt.
#
# The live ISO also contains the union of the target manifests so the desktop
# shown in the live session is the installed desktop, not a mock-up.

arch-install-scripts         # pacstrap, arch-chroot and genfstab for the installer
mkinitcpio-archiso           # archiso hooks for the live initramfs
gum                          # TUI screens
parted                       # inspection and free-space discovery
gptfdisk                     # sgdisk partition editing
jq                           # answers JSON and installer state
pciutils                     # lspci hardware detection
usbutils                     # lsusb diagnostics
vulkan-nouveau               # NVK Vulkan driver so live NVIDIA cards work without the proprietary stack
nano                         # last-resort editing on the live system

# Build-host-only packages are documented in docs/BUILD.md and are not
# installed into the ISO: archiso, devtools, qemu-system-x86, edk2-ovmf,
# squashfs-tools, libisoburn, mtools, shellcheck, shfmt.
